
#include <iostream>

#include <cmath>

using namespace std;



int main() {
    int num;
    long long fac = 1;
    cin >> num;
    int arr[num];

    
    for (int i = 0; i < num;i++){
        cin >> arr[i];
    }
    for (int j = 0; j < num ; j++){
        for (int z = arr[j]; z >=1 ;z--){
            fac = fac * z;
        }
        cout << fac<< endl;
        fac = 1.0;
    }
    return 0;
}
