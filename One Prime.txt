
#include <iostream>

#include <cmath>

using namespace std;



int main() {
    int num;
    cin >> num;
    int counter = 0;
    for (int i = 2; i < num ; i++){
        if (num % i == 0){
            counter ++;
        }
    }
    if (counter >= 1){
        cout << "NO";
    }
    else{
        cout << "YES";
    }
    return 0;
}
