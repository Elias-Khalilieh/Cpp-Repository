
#include <iostream>

#include <cmath>

using namespace std;



int main() {
    char symbol;
    cin >> symbol;
    int number;
    cin >> number;
    
    int arr[number];
    for (int i = 0; i < number; i++){
        cin >> arr[i];
    }
    
    for (int j = 0; j < number; j++){
        for (int k = 0; k < arr[j]; k++){
            cout << symbol;
        }
        cout << endl;
    }
   
}
