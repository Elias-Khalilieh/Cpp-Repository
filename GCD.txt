
#include <iostream>

#include <cmath>

using namespace std;



int main() {
    int A,B,min;
    int GCD = 0;
    cin >> A >> B;
    
    if (A < B){
        min = A;
    }
    else{
        min = B;
    }
    
    for (int i = 1; i <= min ; i++){
        if (A % i == 0 && B % i == 0){
            GCD = i;
        }
    }
    cout << GCD;
   
}
