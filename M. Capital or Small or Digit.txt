
#include <iostream>
using namespace std;


int main() {
    char X;
    int X_to_digit;
    cin >> X;
    
    X_to_digit = int (X);
    if (X >= 48 && X <= 57){
        cout << "IS DIGIT";
    }
    else if (X >=65 && X <=90){
        cout << "ALPHA" <<endl;
        cout << "IS CAPITAL";
    }
    else{
        cout << "ALPHA"<<endl;
        cout << "IS SMALL";
    }
    return 0;
}
