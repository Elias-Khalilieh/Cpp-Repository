
#include <iostream>
using namespace std;
int main() {
    // Write C++ code here
    int A,B;
    char S;
    cin >> A >> S >> B;
    
    if (S == '>'){
        if (A > B==1){
            cout<<"Right";
        }
        else{
            cout <<"Wrong";
        }
        }
    else if (S == '<'){
        if (A<B ==1){
            cout<<"Right";
        }
        else{
            cout <<"Wrong";
        }
    }
    else if (S == '='){
        if (A == B){
            cout<<"Right";
        }
        else{
            cout <<"Wrong";
        }
    }
    return 0;
}
